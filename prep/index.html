function doGet(e) { var p = e.parameter || {}; var sh = SpreadsheetApp.getActive().getSheets()[0]; var data = sh.getDataRange().getValues(); if (p.action === 'set') { var lock = LockService.getScriptLock(); lock.waitLock(5000); for (var i = 1; i < data.length; i++) { if (data[i][0] == p.shop && data[i][1] == p.item) { var max = Number(data[i][4]) || 99; var c = Math.max(0, Math.min(max, Number(p.count) || 0)); sh.getRange(i + 1, 3).setValue(c); sh.getRange(i + 1, 8).setValue(new Date()); } } lock.releaseLock(); data = sh.getDataRange().getValues(); } var rows = []; for (var j = 1; j < data.length; j++) { if (!p.shop || data[j][0] == p.shop) { rows.push({shop: data[j][0], name: data[j][1], count: data[j][2], goal: data[j][3], max: data[j][4], redBelow: data[j][5], order: data[j][6], updatedAt: data[j][7], category: data[j][8]}); } } return ContentService.createTextOutput(JSON.stringify(rows)) .setMimeType(ContentService.MimeType.JSON); }
